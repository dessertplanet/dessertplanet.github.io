{{/* The lead slot: the one image or video that represents a project, rendered between the header block and the body. It lives in front matter rather than in the markdown body because it is metadata, not prose -- which is what lets head.html reuse it as the page's og:image, and what keeps its position identical on every project rather than a per-page decision. It also sits outside .prose, so it is free of the 62ch reading measure that the body is capped at. Three shapes, one field: lead: image: shot.jpg # a file beside index.md in the page bundle alt: "..." # required: this is content, not decoration caption: "..." # optional lead: video: dQw4w9WgXcQ # a YouTube id caption: "..." # optional lead: clip: demo.mp4 # a short silent loop beside index.md alt: "..." # what it shows, as for an image caption: "..." # optional */}} {{- with .Params.lead -}} {{- $page := $.Page -}} {{- $caption := .caption -}} {{- $kinds := 0 -}} {{- range (slice .image .video .clip) -}}{{- if . -}}{{- $kinds = add $kinds 1 -}}{{- end -}}{{- end -}} {{- if gt $kinds 1 -}} {{- errorf "lead: %s sets more than one of image, video and clip -- a project leads with one thing" $page.File.Path -}} {{- end -}} {{- with .video -}} {{- end -}} {{- with .clip -}} {{- $clip := $page.Resources.GetMatch . -}} {{- if not $clip -}} {{- errorf "lead.clip %q not found for %s -- a lead clip has to sit beside its index.md, so this page needs to be a leaf bundle: move it to content/projects//index.md and put the file in that folder" . $page.File.Path -}} {{- else -}} {{- $alt := $.Params.lead.alt | default $caption | default (printf "%s video" $page.Title) -}} {{- end -}} {{- end -}} {{- with .image -}} {{- $img := $page.Resources.GetMatch . -}} {{/* errorf logs and fails the build, but it does not stop this template, so the rendering below sits in the else branch -- left to run against a nil resource it raises a second, uglier nil-pointer error that buries the readable one. */}} {{- if not $img -}} {{- errorf "lead.image %q not found for %s -- a lead image has to sit beside its index.md, so this page needs to be a leaf bundle: move it to content/projects//index.md and put the file in that folder" . $page.File.Path -}} {{- else -}} {{/* Candidate widths for the srcset, capped at the source's own width so a small screenshot is never upscaled into a bigger, blurrier file. The source width itself is always a candidate: without it an image narrower than the next step up (the 972px screenshot against a 1180px step) would only ever be served at 640 and never at its full resolution. */}} {{- $widths := slice -}} {{- range (slice 640 960 1180 1770 2360) -}} {{- if lt . $img.Width -}}{{- $widths = $widths | append . -}}{{- end -}} {{- end -}} {{- $widths = $widths | append $img.Width -}} {{- $srcset := slice -}} {{- range $widths -}} {{- $r := $img.Resize (printf "%dx webp q82" .) -}} {{- $srcset = $srcset | append (printf "%s %dw" $r.RelPermalink .) -}} {{- end -}} {{- $fallback := $img.Resize (printf "%dx jpg q82" $img.Width) -}} {{- end -}} {{- end -}} {{- end -}}
